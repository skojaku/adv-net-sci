% Ringville, and the two other figures the Module 2 sheet draws. Kept in their
% own file because the answer key draws exactly the same town with the boxes
% filled in. Needs tikz, its calc and positioning libraries, and the colours
% pltblue and pltrust, all of which the including file sets up.
% ---------------------------------------------------------------------------
% The town: sixteen people in a circle, everybody friends with the two on
% their left and the two on their right. Four friends each, thirty-two
% friendships. That one drawing has to carry both halves of the module --
% distance AND triangles -- which is why it is next-two-neighbours rather than
% next-one: a ring of next-one neighbours has no triangle anywhere in it.
%
% Two more things fall out of the choice and both are used on the sheet. The
% town is left-right symmetric about the line through person 1 and person 9,
% so a student who has worked out people 2 to 9 already has 10 to 16. And the
% two shortcuts (1--9 and 5--13) are diameters, so they cross at the middle of
% the drawing and read as two lines rather than as a tangle.
% ---------------------------------------------------------------------------
\tikzset{
  % Every friendship is drawn the same, so no line weight says "this one
  % matters more". The two shortcuts are the exception, and they are the point.
  friend/.style   = {line width=1.4pt, black!58, line cap=round},
  shortcut/.style = {line width=2.1pt, pltrust, line cap=round},
  chair/.style    = {circle, draw=black!72, fill=white, line width=0.9pt,
                     minimum size=0.66cm, inner sep=0pt, font=\footnotesize},
  chairhot/.style = {chair, draw=pltblue!75!black, line width=1.5pt,
                     fill=pltblue!14},
  % The box a student writes a handshake count into, one per person, printed
  % outside the ring where no friendship line can reach it.
  dbox/.style     = {draw=pltblue!70, line width=0.7pt, fill=white,
                     minimum size=0.58cm, inner sep=0pt, rounded corners=1.2pt},
  dnum/.style     = {font=\bfseries\footnotesize, inner sep=0pt},
  % A pair that might or might not be a friendship. The student darkens it.
  maybe/.style    = {line width=1.5pt, black!62, dash pattern=on 1.7pt off 2.3pt},
  % The rest of the town in the neighbourhood figure: present, so the pattern
  % can still be read off it, but pushed so far towards white that the eye goes
  % straight to person 3 and the four people around them.
  faint/.style    = {line width=1.2pt, black!11, line cap=round},
  chairfaint/.style = {chair, draw=black!17, text=black!24, line width=0.7pt},
  % Person 3, and the four friendships that make them the subject of Part 3.
  chairme/.style  = {chair, draw=pltrust!65!black, fill=pltrust, text=white,
                     line width=1.2pt, font=\footnotesize\bfseries},
  mine/.style     = {line width=1.7pt, pltrust, line cap=round},
}

% \ringtown{extra lines}{numbers already written in}.
%
% The count goes in the box beside the person, on the drawing, rather than
% into a list of blanks underneath it: the thing the sheet is teaching is that
% the number belongs to a seat, and a column of fifteen blanks hides that.
\newcommand{\ringtown}[2]{%
\begin{tikzpicture}[x=1cm, y=1cm]
    \foreach \i in {1,...,16} {
        \pgfmathsetmacro{\ang}{90 - (\i-1)*22.5}
        \coordinate (c\i) at (\ang:2.7);
    }
    % Next door, both sides: the rim of the circle.
    \foreach \i in {1,...,16} {
        \pgfmathtruncatemacro{\j}{mod(\i,16)+1}
        \draw[friend] (c\i) -- (c\j);
    }
    % One seat over, both sides. Bowed inward: drawn straight, the chord passes
    % 0.14cm inside the chair sitting between its two ends and is drawn through
    % that person's face.
    \foreach \i in {1,...,16} {
        \pgfmathtruncatemacro{\j}{mod(\i+1,16)+1}
        \pgfmathsetmacro{\amid}{90 - \i*22.5}
        \draw[friend] (c\i) .. controls (\amid:2.12) .. (c\j);
    }
    #1
    \foreach \i in {1,...,16} {
        \pgfmathsetmacro{\ang}{90 - (\i-1)*22.5}
        \node[chair] (p\i) at (\ang:2.7) {\i};
        \node[dbox, name=b\i] at (\ang:3.44) {};
    }
    #2
\end{tikzpicture}}

% ---------------------------------------------------------------------------
% Person 3 and their four friends -- the same town, in the same seats, with
% the contrast turned down everywhere else. An earlier version lifted the four
% out into a separate square and nobody could tell what it was a picture of;
% keeping the circle means the student is looking at the object they have been
% counting on for two pages.
%
% Person 3 is red and so are the four friendships that make them the subject.
% The six pairs among those four friends are dashed, and the three of the six
% that really are ring friendships are NOT also drawn as ordinary lines -- a
% solid line under a dashed one would answer the question before it is asked.
% Everything else in the town stays, at black!11, so the pattern (everybody
% knows the two on each side) can still be read off the rest of the circle.
% ---------------------------------------------------------------------------
\newcommand{\hoodfig}[1]{%
\begin{tikzpicture}[x=1cm, y=1cm]
    \foreach \i in {1,...,16} {
        \pgfmathsetmacro{\ang}{90 - (\i-1)*22.5}
        \coordinate (c\i) at (\ang:2.7);
    }
    % The rim, minus the four seats between person 1 and person 5.
    \foreach \i in {5,...,16} {
        \pgfmathtruncatemacro{\j}{mod(\i,16)+1}
        \draw[faint] (c\i) -- (c\j);
    }
    % One seat over, minus 1--3, 2--4 and 3--5.
    \foreach \i in {4,...,16} {
        \pgfmathtruncatemacro{\j}{mod(\i+1,16)+1}
        \pgfmathsetmacro{\amid}{90 - \i*22.5}
        \draw[faint] (c\i) .. controls (\amid:2.12) .. (c\j);
    }
    % The six pairs. All four chords are bowed inward, to a different depth
    % each, so that no two of them run together and none is drawn across a
    % face: 2--4 passes 0.65cm inside person 3, then 1--4 and 2--5, then 1--5
    % innermost. Control radii chosen so the midpoint of each curve sits at
    % 2.05, 1.62, 1.62 and 1.10 from the centre.
    \draw[maybe] (c1) -- (c2);
    \draw[maybe] (c4) -- (c5);
    \draw[maybe] (c2) .. controls (45:1.90)    .. (c4);
    \draw[maybe] (c1) .. controls (56.25:1.41) .. (c4);
    \draw[maybe] (c2) .. controls (33.75:1.41) .. (c5);
    \draw[maybe] (c1) .. controls (45:0.83)    .. (c5);
    #1
    % Person 3's own four friendships.
    \draw[mine] (c2) -- (c3);
    \draw[mine] (c3) -- (c4);
    \draw[mine] (c1) .. controls (67.5:2.12) .. (c3);
    \draw[mine] (c3) .. controls (22.5:2.12) .. (c5);
    \foreach \i in {6,...,16} {
        \pgfmathsetmacro{\ang}{90 - (\i-1)*22.5}
        \node[chairfaint] at (\ang:2.7) {\i};
    }
    \foreach \i in {1,2,4,5} {
        \pgfmathsetmacro{\ang}{90 - (\i-1)*22.5}
        \node[chair] at (\ang:2.7) {\i};
    }
    \node[chairme] at (45:2.7) {3};
\end{tikzpicture}}

    \foreach \i in {1,2,3} {
        \draw[twig] (r) -- (a\i);
        \foreach \j in {1,2,3} { \draw[twig] (a\i) -- (a\i b\j); }
    }

        \node[tdot] at (a\i) {};
        \foreach \j in {1,2,3} { \node[tdot] at (a\i b\j) {}; }
    }
    \node[tme] at (r) {};
    \node[tlab] at (0,-1.70)   {you};
    \node[tlab] at (1.6,-1.70) {150};
    \node[tlab] at (3.2,-1.70) {$150\times150$};
